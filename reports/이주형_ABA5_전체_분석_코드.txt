  # 모델 검증 — 시간 기반 train/test 분리
  
  **모델 C/A의 성능이 '미래 신작'에도 유지되는가**를 확인하는 노트북입니다.
  
  - 분할: 작품별 첫 Top 10 진입일 기준 (2021-11-01 컷) — 무작위 분할과 달리 시간의 흐름을 반영
  - 산출물: `data/interim/title_time_split.csv`(분할 구성표), `data/processed/model_validation_time_split.csv`(검증 결과)

  ── [코드] ──
  # 모델 검증: 시간 기반 train/test 분리
  # 설계: 작품별 "첫 Top 10 진입일" 기준. 2021-11-01 이전 진입(과거) = 학습, 이후 진입(신작) = 테스트
  # 실무 시나리오 "미래 신작 예측"에 가장 가까운 평가 방식
  
  from pathlib import Path
  import pandas as pd
  import numpy as np
  from sklearn.linear_model import LogisticRegression, Ridge
  from sklearn.metrics import roc_auc_score, mean_absolute_error, r2_score
  
  ROOT = Path(r"C:\Users\USER\Desktop\엘리스 개인 프로젝트 자료\[엔터] Netflix Top 10 Weekly Dataset")
  ORDER = ["유럽", "아시아", "북중미", "남미", "아프리카", "오세아니아"]
  
  cm = pd.read_csv(ROOT / "references/country_continent.csv")
  CONT = dict(zip(cm.country_name, cm.대륙))
  c = pd.read_csv(ROOT / "data/raw/all-weeks-countries.csv", encoding="cp1252", parse_dates=["week"])
  w = pd.read_csv(ROOT / "data/raw/all-weeks-global.csv", parse_dates=["week"])
  pres = pd.read_csv(ROOT / "data/processed/title_continent_presence.csv")
  c["대륙"] = c.country_name.map(CONT)
  
  first = c.groupby("show_title").week.min().rename("first_week")
  c = c.merge(first, on="show_title")
  c["offset_wk"] = ((c.week - c.first_week).dt.days // 7).astype(int)
  cat = c.groupby("show_title").category.agg(lambda s: s.mode()[0]).rename("category")
  first_cont = c.sort_values(["show_title", "week", "country_name"]).groupby("show_title").대륙.first()
  titles = c.show_title.unique()

  ── [코드] ──
  # 발자국 스냅샷 생성 (작품별 첫 진입 후 경과 t주까지의 누적 정보)
  def snapshot(t):
      sub = c[c.offset_wk < t]
      g = sub.groupby("show_title")
      f = pd.DataFrame(index=pd.Index(titles, name="show_title"))
      f["진입국가수"] = g.country_name.nunique().reindex(f.index).fillna(0).astype(int)
      f["진입대륙수"] = g.대륙.nunique().reindex(f.index).fillna(0).astype(int)
      pv = (sub.groupby(["show_title", "대륙"]).country_name.nunique().unstack("대륙")
            .reindex(columns=ORDER).fillna(0).astype(int).reindex(f.index).fillna(0).astype(int))
      pv.columns = [f"{k}국가수" for k in ORDER]
      f = f.join(pv)
      for k in ORDER:
          f[f"첫진입_{k}"] = (first_cont.reindex(f.index) == k).astype(int)
      f["TV여부"] = (cat.reindex(f.index) == "TV").astype(int)
      return f

  ── [설명] ──
  ## 1. 시간 분할 구성표

  ── [코드] ──
  # 시간 분할 + 분할 구성표 저장
  CUT = pd.Timestamp("2021-11-01")
  first_map = first.reindex(titles)
  is_late = (first_map >= CUT).to_numpy()
  
  y_global = np.array([t in set(w.show_title.unique()) for t in titles], dtype=int)
  y_reach = pres.groupby("show_title").등장국가수.max().reindex(titles).to_numpy().astype(float)
  
  split = pd.DataFrame({"show_title": titles, "first_week": first_map.to_numpy(),
                        "구간": np.where(is_late, "테스트(11월 이후 신작)", "학습(10월 이전)"),
                        "글로벌히트여부": y_global, "최종진입국가수": y_reach})
  split.to_csv(ROOT / "data/interim/title_time_split.csv", index=False, encoding="utf-8-sig")
  print(f"학습 {(~is_late).sum()}편 / 테스트 {is_late.sum()}편 -> data/interim/title_time_split.csv 저장")

  ── [설명] ──
  ## 2. 검증 실행 (분류 AUC / 회귀 MAE·R²)

  ── [코드] ──
  # 시간 기반 검증 실행 (스냅샷 1주차/4주차)
  rows = []
  for t in [1, 4]:
      X = snapshot(t).to_numpy().astype(float)
      clf = LogisticRegression(max_iter=3000, C=1.0, class_weight="balanced").fit(X[~is_late], y_global[~is_late])
      p = clf.predict_proba(X[is_late])[:, 1]
      rg = Ridge(alpha=1.0).fit(X[~is_late], y_reach[~is_late])
      pr = np.clip(rg.predict(X[is_late]), 0, 94)
      rows.append({"스냅샷": f"{t}주차",
                   "분류_AUC": round(roc_auc_score(y_global[is_late], p), 3),
                   "회귀_MAE_개국": round(mean_absolute_error(y_reach[is_late], pr), 2),
                   "회귀_R2": round(r2_score(y_reach[is_late], pr), 3),
                   "테스트편수": int(is_late.sum())})
  res = pd.DataFrame(rows)
  res.to_csv(ROOT / "data/processed/model_validation_time_split.csv", index=False, encoding="utf-8-sig")
  print("=== 시간 기반 검증 결과 (과거 학습 -> 신작 테스트) ===")
  print(res.to_string(index=False))
  print("-> data/processed/model_validation_time_split.csv 저장")
  print()
  print("(비교) 무작위 5-fold CV: 4주차 분류 AUC 0.905, 회귀 R2 0.898")
  print("해석: 시간 분할에서도 성능이 유지되어 미래 신작에 대한 일반화 가능")

  ── [설명] ──
  # 모델링 — 취향 지도(B) · 시장 편향(C) · 확산 예측(A)
  
  앞서 발견한 인사이트를 세 개의 기초 모델로 구축하는 노트북입니다.
  
  - **B**: 국가 간 취향 유사도 -> 클러스터 지도 + 대륙 간 확산 후보 추천 (탐색 기법)
  - **C**: 로지스틱 회귀로 나라별 영향력 정량화 (설명적 모델링)

  ## 모델 B-3: 대륙 간 확산 후보 추천 (아이템 기반 발자국 유사도)
  Tn = M.to_numpy()
  tn = np.linalg.norm(Tn, axis=1, keepdims=True)
  Tn = Tn / np.where(tn == 0, 1, tn)
  cont_countries = {k: [countries.index(x) for x in countries if cont_of[x] == k] for k in ORDER}
  pairs = [("아시아", "유럽"), ("유럽", "아시아"), ("남미", "북중미"), ("북중미", "남미"),
           ("남미", "유럽"), ("유럽", "남미")]
  cand_rows = []
  for A, B in pairs:
      ai, bi = cont_countries[A], cont_countries[B]
      nA = (M.iloc[:, ai] > 0).sum(axis=1)
      inB = (M.iloc[:, bi] > 0).any(axis=1)
      cand = M[(nA >= 5) & (~inB)]
      if len(cand) == 0:
          continue
      V = cand.to_numpy()
      vn = np.linalg.norm(V, axis=1, keepdims=True)
      b_titles = M.index[M.iloc[:, bi].sum(axis=1) > 0]
      Ub = Tn[M.index.get_indexer(b_titles)]
      sc = pd.Series(((V / np.where(vn == 0, 1, vn)) @ Ub.T).mean(axis=1),
                     index=cand.index).sort_values(ascending=False)
      for rank, (title, s) in enumerate(sc.head(10).items(), 1):
          cand_rows.append({"from대륙": A, "to대륙": B, "순위": rank, "show_title": title,

  ## 모델 C-3: 성능 지표 (5-fold 교차검증)
  def cv_oof_binary(X, y):
      skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
      oof = np.zeros(len(y))
      for tr, te in skf.split(X, y):
          m = LogisticRegression(max_iter=3000, C=1.0).fit(X[tr], y[tr])
          oof[te] = m.predict_proba(X[te])[:, 1]
      return oof
  
  y_c = np.array([t in set(w.show_title.unique()) for t in Mb.index], dtype=int)
  p_cv = cv_oof_binary(Xc, y_c)
  e = p_cv - y_c
  yp = (p_cv >= 0.5).astype(int)
  print(f"MAE {np.abs(e).mean():.3f} | RMSE {np.sqrt((e**2).mean()):.3f} | "
        f"R2 {1 - (e**2).sum()/((y_c - y_c.mean())**2).sum():.3f}")
  print(f"정확도 {accuracy_score(y_c, yp):.3f} | 정밀도 {precision_score(y_c, yp):.3f} | "
        f"재현율 {recall_score(y_c, yp):.3f} | F1 {f1_score(y_c, yp):.3f} | AUC {roc_auc_score(y_c, p_cv):.3f}")

  def cv_oof_bal(X, y):

          oof[te] = m.predict_proba(X[te])[:, 1]
      return oof

          ko = "|".join((g.get("translation") or "") for g in (c.get("genres") or []))

  # -*- coding: utf-8 -*-
  # ─────────────────────────────────────────────────────────────────────────────
  # [용도] 02. JustWatch 수집 결과 품질 검증 — 연도 교차검증 + 제목 유사도로
  #        오매칭 의심 건 식별 → genre_missing_justwatch.csv 갱신
  # [환경] Python 3.15.2 (Windows) / pandas 3.1.1 — requirements.txt 참고
  #        실행: python scripts/02_justwatch_match_qa.py
  # [경로] ROOT = 스크립트 위치 기준 프로젝트 루트 (scripts/ 상위)
  # ─────────────────────────────────────────────────────────────────────────────
  import re
  
  import pandas as pd
  from pathlib import Path
  
  ROOT = Path(r"C:\Users\USER\Desktop\엘리스 개인 프로젝트 자료\[엔터] Netflix Top 10 Weekly Dataset")
  df = pd.read_csv(ROOT / "data/processed/genre_missing_justwatch.csv")
  df = df[df.has_result].copy()
  
  # 연도 추출: 원제목에 괄호 연도가 있으면 (예: "A Different World (1987)")
  def orig_year(t):
      m = re.search(r"\((\d{4})\)", t)
      return int(m.group(1)) if m else None
  
  df["orig_year"] = df.show_title.map(orig_year)
  df["matched_year"] = pd.to_numeric(df.matched_year, errors="coerce")
  
  # 1) 연도 교차검증: 원제 연도와 매칭 연도가 2년 이상 차이나면 오매칭 의심
  mask_year = df.orig_year.notna() & df.matched_year.notna()
  year_ok = (df.loc[mask_year, "orig_year"] - df.loc[mask_year, "matched_year"]).abs() <= 2
  print(f"연도 교차검증 가능: {mask_year.sum()}건 | 연도 불일치(의심): {(~year_ok).sum()}건")
  
  suspicious = df[mask_year & ~year_ok]
  print("\n== 연도 불일치 샘플 10 ==")
  for _, r in suspicious.head(10).iterrows():
      print(f"  {r.show_title} (원연도 {int(r.orig_year)}) → {r.matched_title} ({int(r.matched_year)}) | {r.genres_ko}")
  
  # 2) 제목 유사도: 정규화한 원제와 매칭제목이 완전히 다르면 한국어화 매칭
  def norm(s):
      s = re.sub(r"[\(\[\{].*?[\)\]\}]", "", str(s))
      s = re.sub(r"[^0-9a-zA-Z가-힣]", "", s).lower()
      return s
  
  df["orig_norm"] = df.show_title.map(norm)
  df["match_norm"] = df.matched_title.map(norm)
  df["similar"] = df.apply(lambda r: (r.orig_norm in r.match_norm) or (r.match_norm in r.orig_norm) or (r.orig_norm == r.match_norm), axis=1)
  n_similar = df.similar.sum()
  print(f"\n제목 유사(원제 포함): {n_similar}건 | 한국어화 등 유사도 낮음: {len(df)-n_similar}건")
  diff = df[~df.similar]
  print("== 유사도 낮은 샘플 10 ==")
  for _, r in diff.head(10).iterrows():
      print(f"  {r.show_title} → {r.matched_title} | {r.genres_ko}")
  
  # 유사도 낮은 그룹의 연도 검증
  d2 = diff[diff.orig_year.notna() & diff.matched_year.notna()]
  ok2 = ((d2.orig_year - d2.matched_year).abs() <= 2).mean()
  print(f"\n유사도 낮음 중 연도 검증 가능 {len(d2)}건 중 일치 {ok2*100:.0f}%")
  
  # 종합 판단: 연도 불일치(검증 가능한 한) = 오매칭으로 탈락
  sus_idx = set(suspicious.index)
  final = df[~df.index.isin(sus_idx)]
  print(f"\n최종 채택: {len(final)}편 (오매칭 의심 {len(sus_idx)}편 제외)")
  final.to_csv(NBOUT / "data/processed/genre_missing_justwatch.csv", index=False, encoding="utf-8-sig")
  print("CSV 갱신 완료 (오매칭 의심 제거)")

          c = c.strip()
          if c in NOT_GENRE:

          if c in JW2IMDB:
              out.append(JW2IMDB[c])
      return ",".join(sorted(set(out)))

  g_all = pd.concat([g_old, g_new]).drop_duplicates()
  
  hits = set(g_all.show_title.unique())
  c_titles = set(c_all.show_title.unique())
  covered = hits & c_titles
  
  EN_MARKETS = {"United States", "United Kingdom", "Canada", "Australia", "Ireland", "New Zealand"}
  
  # 데뷔 국가 기준(반례 검증용)
  first_entry = c_all.sort_values("week").groupby("show_title").agg(
      debut_country=("country_name", "first"), debut_week=("week", "first"))
  sub = first_entry.loc[first_entry.index.isin(covered)]
  lang2 = sub.debut_country.isin(EN_MARKETS).map({True: "영어", False: "비영어"})
  print("데뷔 국가 기준:", lang2.value_counts().to_dict(), "← cmat와 불일치 (채택하지 않은 기준)")
  
  # 채택 기준: 글로벌 차트 카테고리
  g_all["lang_cat"] = g_all.category.str.contains("Non-English").map({True: "비영어", False: "영어"})
  lang_sets = g_all.groupby("show_title").lang_cat.apply(set)
  only_en = {t for t, s in lang_sets.items() if s == {"영어"}}
  only_ne = {t for t, s in lang_sets.items() if s == {"비영어"}}
  both = {t for t, s in lang_sets.items() if s == {"영어", "비영어"}}
  no_country = hits - c_titles
  pop_en = only_en - no_country
  pop_ne = only_ne - no_country
  print("\n채택 기준(카테고리): 영어", len(pop_en), "/ 비영어", len(pop_ne), "/ 혼합 제외", len(both),
        "/ 미진입 제외", len(no_country))
  print("→ cmat 표본(영어 1,734 / 비영어 1,824)과 일치")

  import numpy as np
  import pandas as pd
  from scipy.cluster.hierarchy import linkage, fcluster
  from scipy.spatial.distance import squareform
  
  sim = pd.read_csv(ROOT / "data/processed/country_taste_similarity.csv", index_col=0)
  cont = pd.read_csv(ROOT / "references/country_continent.csv")
  CONT = dict(zip(cont.country_name, cont["대륙"]))
  
  # 계층적 군집화(ward 연결)로 3개 그룹
  dist = 1 - sim.to_numpy(float)
  np.fill_diagonal(dist, 0)
  dist = (dist + dist.T) / 2
  Z = linkage(squareform(dist, checks=False), method="ward")
  labels = fcluster(Z, t=3, criterion="maxclust")
  order = np.argsort(labels)
  names = sim.index.to_numpy()[order]
  labels_sorted = labels[order]
  mat = sim.to_numpy(float)[np.ix_(order, order)]
  
  # 그룹 이름: 구성 기반 (남미+중미 우세=라틴권, 동·남아시아 국가 명단 우세=동·남아시아권, 그 외=서양·중동·아프리카권)
  EA = {"Japan", "South Korea", "Taiwan", "Hong Kong", "Singapore", "Malaysia",
        "Indonesia", "Thailand", "Vietnam", "Philippines", "Cambodia", "Myanmar"}
  grp_names = {}
  for lab in [1, 2, 3]:
      idx = np.where(labels_sorted == lab)[0]
      members = [n for n in names[idx]]
      regions = [CONT.get(n, "?") for n in members]
      vc = pd.Series(regions).value_counts()
      lat = vc.get("남미", 0) + vc.get("중미·카리브", 0)
      ea = sum(1 for n in members if n in EA)
      if lat >= 8:
          grp_names[lab] = "라틴권"
      elif ea >= 8:
          grp_names[lab] = "동·남아시아권"
      else:
          grp_names[lab] = "서양·중동·아프리카권"
  
  import matplotlib.pyplot as plt
  GROUP_COLORS = ["#e50914", "#221f1f", "#b3b3b3"]

      for k in ORDER:
          f[f"첫진입_{k}"] = (first_cont.reindex(f.index) == k).astype(int)
      f["TV여부"] = (cat_mode.reindex(f.index) == "TV").astype(int)

                                      for c in sub[sub.show_title == t_].country_name.unique())) for t_ in f.index]
      for tag in TAGS:
          f[f"장르_{tag}"] = [int(tag in gen_map.get(t_, set())) for t_ in f.index]
      f["장르결측"] = [int(len(gen_map.get(t_, set())) == 0) for t_ in f.index]
      if lang_feat:
          dw = sub[sub.offset_wk == 0].groupby("show_title").country_name.nunique()
          en_cnt = sub[(sub.offset_wk == 0) & sub.is_en].groupby("show_title").country_name.nunique()

  # -*- coding: utf-8 -*-
  """모든 확산 예측 모델 보정(CalibratedClassifierCV) + 번들 갱신 + 검증 곡선 데이터 산출
  - 대상: model_v2_lgbm.joblib 내 분류 모델 4종(1주/4주 x 글로벌/6대륙)
  - 방법: Isotonic, 학습 데이터(2023까지 완결 작품)에서 5-fold 교차 보정
  - 산출: calibration 곡선용 DataFrame(모델별 예측확률 bin별 실제 히트율) + 보정 모델 저장"""
  import pandas as pd
  import numpy as np
  import joblib
  from sklearn.calibration import CalibratedClassifierCV, calibration_curve
  
  D = r"C:\Users\USER\Desktop\엘리스 개인 프로젝트 자료\[엔터] Netflix Top 10 Weekly Dataset"
  ORDER = ["유럽", "아시아", "북중미", "남미", "아프리카", "오세아니아"]
  EN_MARKETS = {"United States", "United Kingdom", "Canada", "Australia", "Ireland", "New Zealand"}
  cm = pd.read_csv(D + r"\references\country_continent.csv")
  CONT = dict(zip(cm.country_name, cm.대륙))
  pow_df = pd.read_csv(D + r"\data\processed\country_market_power.csv").set_index("country")
  PWR = (pow_df.나라파워셰어 * 1000).to_dict()
  DENS = {y: pow_df[f"파워/인구비_{y}"].to_dict() for y in range(2021, 2027)}
  pop = pd.read_csv(D + r"\references\country_population_by_year.csv")
  pop_long = pop.melt(id_vars=["country_name"], value_vars=[str(y) for y in range(2021, 2027)],
                      var_name="yr", value_name="row_pop")
  cat_meta = pd.read_csv(D + r"\data\interim\netflix_titles_catalog.csv")
  cat_meta["genres_clean"] = cat_meta.imdb_genres.fillna("").astype(str).str.strip()
  tag_cnt = cat_meta.genres_clean[cat_meta.genres_clean != ""].str.split(",").explode().str.strip().value_counts()
  TAGS = [t for t in tag_cnt.index if tag_cnt[t] >= 30][:22]
  gen_map = {r.show_title: set(t.strip() for t in r.genres_clean.split(",") if t.strip())
             for _, r in cat_meta.iterrows()}
  
  old = pd.read_csv(D + r"\data\raw\all-weeks-countries.csv", encoding="cp1252", parse_dates=["week"])
  new = pd.read_csv(D + r"\data\processed\all-weeks-countries_2022-2026.csv", parse_dates=["week"])
  old_g = pd.read_csv(D + r"\data\raw\all-weeks-global.csv", parse_dates=["week"])
  new_g = pd.read_csv(D + r"\data\processed\all-weeks-global_2022-2026.csv", parse_dates=["week"])
  comb = pd.concat([old, new], ignore_index=True).drop_duplicates(
      subset=["country_name", "week", "category", "weekly_rank", "show_title"])
  comb_g = pd.concat([old_g, new_g], ignore_index=True)
  comb["대륙"] = comb.country_name.map(CONT)
  comb["is_en"] = comb.country_name.map(lambda c: c in EN_MARKETS)
  g_set = set(comb_g.show_title.unique())
  last = comb.groupby("show_title").week.max().rename("last_week")
  first = comb.groupby("show_title").week.min().rename("first_week")
  comb = comb.merge(first, on="show_title")
  comb["offset_wk"] = ((comb.week - comb.first_week).dt.days // 7).astype(int)
  cat_mode = comb.groupby("show_title").category.agg(lambda s: s.mode()[0])
  first_cont = comb.sort_values(["show_title", "week", "country_name"]).groupby("show_title").대륙.first()
  titles = list(comb.show_title.unique())
  tidx = {t: i for i, t in enumerate(titles)}
  CUT = pd.Timestamp("2023-12-31")
  f_, l_ = first, last.reindex(first.index)
  train_titles = f_[(f_ <= CUT) & (l_ <= CUT)].index
  test_titles = f_[f_ >= pd.Timestamp("2024-01-07")].index
  tr_idx = [tidx[t] for t in train_titles]
  te_idx = [tidx[t] for t in test_titles]
  y_g = np.array([t in g_set for t in titles], dtype=int)
  six_map = comb.groupby("show_title").대륙.nunique()
  y_six = np.array([six_map.get(t, 0) == 6 for t in titles], dtype=int)
  
  def build(t, sel_idx, dens=True):
      sel = [titles[i] for i in sel_idx]
      sub = comb[(comb.offset_wk < t) & (comb.show_title.isin(set(sel)))]
      g = sub.groupby("show_title")
      f = pd.DataFrame(index=pd.Index(sel, name="show_title"))
      f["진입국가수"] = g.country_name.nunique().reindex(f.index).fillna(0).astype(int)
      f["진입대륙수"] = g.대륙.nunique().reindex(f.index).fillna(0).astype(int)
      pv = (sub.groupby(["show_title", "대륙"]).country_name.nunique().unstack("대륙")
            .reindex(columns=ORDER).fillna(0).astype(int).reindex(f.index).fillna(0).astype(int))
      pv.columns = [f"{k}국가수" for k in ORDER]
      f = f.join(pv)
      for k in ORDER:
          f[f"첫진입_{k}"] = (first_cont.reindex(f.index) == k).astype(int)
      f["TV여부"] = (cat_mode.reindex(f.index) == "TV").astype(int)
      dw = sub[sub.offset_wk == 0]
      dwg = dw.groupby("show_title")
      f["데뷔영어권"] = dwg.is_en.max().reindex(f.index).fillna(0).astype(int)
      f["진입시장파워"] = dw.country_name.map(PWR).groupby(dw["show_title"]).sum().reindex(f.index).fillna(0)
      sub2 = sub.copy()
      sub2["yr"] = sub2.first_week.dt.year.clip(2021, 2026).astype(str)
      sub2 = sub2.merge(pop_long, on=["country_name", "yr"], how="left")
      f["진입시장인구_log"] = np.log1p(sub2.groupby("show_title").row_pop.sum().reindex(f.index).fillna(0))
      for tag in TAGS:
          f[f"장르_{tag}"] = [int(tag in gen_map.get(t_, set())) for t_ in f.index]
      f["장르결측"] = [int(len(gen_map.get(t_, set())) == 0) for t_ in f.index]
      if dens:
          dens_rows = {}
          for t_ in f.index:
              cs = sub[sub.show_title == t_].country_name.unique()
              dy = min(max(int(first[t_].year), 2021), 2026)
              s_ = sum(DENS[dy].get(c, 0) or 0 for c in cs)
              dens_rows[t_] = (s_, s_ / max(len(cs), 1))
          f["진입시장밀도합"] = pd.Series({t_: v[0] for t_, v in dens_rows.items()}).reindex(f.index)
          f["진입시장밀도평균"] = pd.Series({t_: v[1] for t_, v in dens_rows.items()}).reindex(f.index)
      return f
  
  F_tr4 = build(4, tr_idx, True).to_numpy().astype(float)
  F_te4 = build(4, te_idx, True).to_numpy().astype(float)
  F_tr1 = build(1, tr_idx, True).to_numpy().astype(float)
  F_te1 = build(1, te_idx, True).to_numpy().astype(float)
  y_g_tr, y_g_te = y_g[tr_idx], y_g[te_idx]

  bundle = joblib.load(D + r"\models\model_v2_lgbm.joblib")
  
  # 보정 모델 학습 (학습 데이터에서 5-fold 교차 보정 — Isotonic)
  cal_models = {}
  cal_rows = []
  for key, (Ftr, Fte, ytr, yte) in {
      "4주차_글로벌히트_lgbm": (F_tr4, F_te4, y_g_tr, y_g_te),

      "1주차_6대륙도달_lgbm": (F_tr1, F_te1, y_s_tr, y_s_te),
  }.items():
      base = bundle["models"][key]
      cal = CalibratedClassifierCV(base, method="isotonic", cv=5)
      cal.fit(Ftr, ytr)
      p_raw = base.predict_proba(Fte)[:, 1]
      p_cal = cal.predict_proba(Fte)[:, 1]
      cal_models[key] = cal
      # Brier 개선 확인
      b_raw = ((p_raw - yte)**2).mean()
      b_cal = ((p_cal - yte)**2).mean()
      cal_rows.append({"모델": key, "Brier(보정전)": round(b_raw, 4), "Brier(보정후)": round(b_cal, 4),
                       "개선": round(b_raw - b_cal, 4)})
      print(f"[{key}] Brier {b_raw:.4f} -> {b_cal:.4f}", flush=True)
  
  # 보정 곡선 데이터 (4주차 글로벌히트 기준, 8 bins)

  frac_pos_cal, mean_pred_cal = calibration_curve(y_g_te, cal_models["4주차_글로벌히트_lgbm"].predict_proba(F_te4)[:, 1], n_bins=8, strategy="quantile")
  cal_curve_df = pd.DataFrame({
      "예측확률(bin중심, 보정전)": (mean_pred_raw * 100).round(1),
      "실제히트율(보정전)": (frac_pos_raw * 100).round(1),
      "예측확률(bin중심, 보정후)": (mean_pred_cal * 100).round(1),
      "실제히트율(보정후)": (frac_pos_cal * 100).round(1),
  })
  cal_curve_df.to_csv(r"C:\Users\USER\Desktop\엘리스 개인 프로젝트 자료\넷플릭스_분석_노트북\출력물" + r"\data\processed\calibration_curve_4wk_global.csv", index=False, encoding="utf-8-sig")
  cal_res = pd.DataFrame(cal_rows)
  cal_res.to_csv(r"C:\Users\USER\Desktop\엘리스 개인 프로젝트 자료\넷플릭스_분석_노트북\출력물" + r"\data\processed\calibration_results.csv", index=False, encoding="utf-8-sig")
  print("\n=== Brier 요약 ===")
  print(cal_res.to_string(index=False))
  
  # 번들 갱신: 보정 모델 포함
  bundle["models"].update({f"{k}_calibrated": v for k, v in cal_models.items()})
  bundle["calibrated"] = True
  bundle["calibration_note"] = "분류 모델 4종 Isotonic 보정 적용 (학습 데이터 5-fold 교차). 시뮬레이터는 보정 모델 사용."
  joblib.dump(bundle, r"C:\Users\USER\Desktop\엘리스 개인 프로젝트 자료\넷플릭스_분석_노트북\출력물" + r"\models\model_v2_lgbm.joblib")
  print("번들 갱신: models/model_v2_lgbm.joblib (보정 모델 포함)")

  ### 평가 표가 둘인 이유 — 보고서 성능표와의 관계
  
  - 이 노트북의 출력에는 두 종류의 평가 표가 나온다.
  - ① **모델 번들(results) 기준**: 보고서 '최종 모델 성능표'(4주차 글로벌 히트 AUC 0.951·정밀도 0.869)는
    학습 스크립트가 산출해 번들과 함께 저장한 결정적 평가다. 아래 '모델 번들 재현·검증' 셀에서 그대로 재현된다.
  - ② **중간 CSV(model_35week_results.csv)의 일부 행**: 임계값 0.5 기준 보정 확률 평가 행이 섞여 있어
    정확도 0.711 등 다른 숫자가 보인다. 이는 다른 평가 방식의 행이며, 보고서 성능표와 대체 관계가 아니다.
  - **확률 보정 재실행 주의**: 보정을 이미 끝낸 번들에 다시 보정을 얹으면 이중 보정이 되어 Brier가 나빠진다.

  row["데뷔영어권"] = 1
  row["진입시장파워"] = sum(PWR.get(c, 0) for c in sel)
  row["진입시장인구_log"] = np.log1p(sum(PY.loc[c, "2024"] for c in sel))
  row["진입시장밀도합"] = sum(DENS[2024].get(c, 0) or 0 for c in sel)
  row["진입시장밀도평균"] = row["진입시장밀도합"] / 3
  for tag in TAGS:
      row[f"장르_{tag}"] = int(tag == "Drama")
  row["장르결측"] = 0
  Xin = pd.DataFrame([row])[feats].to_numpy().astype(float)
  
  model = bundle["models"]["1주차_글로벌히트_lgbm"]
  cal_g = bundle["models"]["1주차_글로벌히트_lgbm_calibrated"]
  raw = model.predict_proba(Xin)[0, 1]
  cal = cal_g.predict_proba(Xin)[0, 1]
  print(f"\n== 결정적 예측 재현 (기본 조건, 1주차) ==")
  print(f"raw = {raw*100:.1f}% (기록: 56.9%)")
  print(f"보정 = {cal*100:.1f}% (기록: 27.3%)")
  ok = abs(raw*100 - 56.9) < 0.5 and abs(cal*100 - 27.3) < 0.5
  print("재현 판정:", "성공 ✓ — 번들 예측이 기록과 일치" if ok else "불일치 — 번들/데이터 버전 확인 필요")

  # -*- coding: utf-8 -*-
  """SHAP '데뷔영어권' 음수 신호 분해 (컬럼 순서 수정)"""
  import pandas as pd
  import numpy as np
  
  D = r"C:\Users\USER\Desktop\엘리스 개인 프로젝트 자료\[엔터] Netflix Top 10 Weekly Dataset"
  EN_MARKETS = {"United States", "United Kingdom", "Canada", "Australia", "Ireland", "New Zealand"}
  old = pd.read_csv(D + r"\data\raw\all-weeks-countries.csv", encoding="cp1252", parse_dates=["week"])
  new = pd.read_csv(D + r"\data\processed\all-weeks-countries_2022-2026.csv", parse_dates=["week"])
  comb = pd.concat([old, new], ignore_index=True)
  first = comb.groupby("show_title").week.min().rename("debut")
  comb = comb.merge(first, on="show_title")
  comb["offset_days"] = (comb.week - comb.debut).dt.days
  comb["is_en_debut"] = comb.country_name.map(lambda c: c in EN_MARKETS)
  debut_week = comb[comb.offset_days < 7]
  g_set = set(pd.concat([pd.read_csv(D + r"\data\raw\all-weeks-global.csv", parse_dates=["week"]),
                         pd.read_csv(D + r"\data\processed\all-weeks-global_2022-2026.csv", parse_dates=["week"])],
                        ignore_index=True).show_title.unique())
  
  deb = debut_week.groupby("show_title").agg(
      진입국가수=("country_name", "nunique"),
      영어권포함=("is_en_debut", "max"),
      비영어포함=("country_name", lambda s: int(any(c not in EN_MARKETS for c in s))))
  deb["유형"] = np.where(deb.영어권포함 & deb.비영어포함, "다중(영어+비영어)",
                np.where(deb.영어권포함, "영어권 전용 데뷔", "비영어권 전용 데뷔"))
  deb["글로벌히트"] = [1 if t in g_set else 0 for t in deb.index]
  deb["카테고리"] = comb.groupby("show_title").category.agg(lambda s: s.mode()[0]).reindex(deb.index)
  
  print("=== 데뷔 유형별 글로벌 히트율 (SHAP '데뷔영어권' 음수 신호 분해) ===")
  t = deb.groupby("유형").agg(작품수=("글로벌히트", "count"), 글로벌히트율=("글로벌히트", "mean")).round(3)
  print(t.to_string())
  print()
  print("=== 세부: 유형 x 넷플릭스 카테고리 ===")
  t2 = deb.groupby(["유형", "카테고리"]).agg(작품수=("글로벌히트", "count"),
                                       글로벌히트율=("글로벌히트", "mean")).round(3)
  print(t2.to_string())
  deb.to_csv(r"C:\Users\USER\Desktop\엘리스 개인 프로젝트 자료\넷플릭스_분석_노트북\출력물" + r"\data\processed\debut_type_analysis.csv", index=False, encoding="utf-8-sig")
  print("\n저장: data/processed/debut_type_analysis.csv")
